// ──────────────────────────────────────────── // API HTTP // ──────────────────────────────────────────── // Esta página se sirve estática desde GitHub Pages, así que no existe el // objeto google.script.run que Apps Script inyecta cuando sirve su propio // HtmlService. Lo reconstruimos acá encima de fetch(), hablando con el // Web App de Apps Script (que ahora solo actúa como API JSON vía doPost). // El resto del código sigue llamando a google.script.run.xxx(...) tal // cual estaba escrito — no hubo que tocar ningún call site. // URL de implementación del Web App de Apps Script (termina en /exec). // Se actualiza cada vez que se crea una NUEVA implementación (Implementar // → Administrar implementaciones). Ver README para el paso a paso. var API_URL = 'PEGAR_AQUI_LA_URL_DE_IMPLEMENTACION/exec'; var API_FUNCTIONS = [ 'hasUsers', 'login', 'logout', 'getSession', 'createInitialAdmin', 'listUsers', 'createUser', 'updateUser', 'deleteUser', 'listClients', 'createClient', 'updateClient', 'getClientGlobalTotal', 'deleteClient', 'processImage', 'uploadImageOnly', 'saveOrder', 'getOrders', 'updateOrder', 'deleteOrder' ]; function apiRunner() { var successHandler = null; var failureHandler = null; var runner = { withSuccessHandler: function(cb) { successHandler = cb; return runner; }, withFailureHandler: function(cb) { failureHandler = cb; return runner; } }; API_FUNCTIONS.forEach(function(fn) { runner[fn] = function() { var args = Array.prototype.slice.call(arguments); // Sin content-type explícito: el body string cae en text/plain, // que es "simple request" para CORS y evita el preflight OPTIONS // (Apps Script no implementa doOptions). fetch(API_URL, { method: 'POST', body: JSON.stringify({ fn: fn, args: args }) }) .then(function(res) { if (!res.ok) throw new Error('Error de red (' + res.status + ')'); return res.json(); }) .then(function(payload) { if (payload.error) throw new Error(payload.error); if (successHandler) successHandler(payload.result); }) .catch(function(err) { if (failureHandler) failureHandler(err); }); }; }); return runner; } // Namespace compatible con google.script.run: cada acceso crea un runner // nuevo (igual que el original), así llamadas concurrentes no se pisan // los handlers entre sí. window.google = window.google || {}; google.script = google.script || {}; Object.defineProperty(google.script, 'run', { get: apiRunner, configurable: true });
